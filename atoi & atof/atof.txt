#include<stdio.h>
int main()
{
	char s[20];
	float num1=0,num2=0,d=1;
	int i=0,flag=1;
	printf("Enter string: ");
	scanf("%s",s);
	if (s[0]=='+')
	{
		flag=1;
		i=1;
	}
	else if(s[0]=='-')
	{
		flag=-1;
		i=1;
	}
	// Before decimal point
	for (;s[i]>='0'&&s[i]<='9';i++)
	{
		num1=num1*10+(s[i]-'0');
	}
	// After decimal point
	if(s[i]=='.')
	{
		i++;
		for (;s[i]>='0'&&s[i]<='9';i++)
		{
			num2=num2*10+(s[i]-'0');
			d=d*10;
		}
	}
	num1=num1+num2/d;
	num1=num1*flag;
	printf("num1=%f\n",num1);
	return 0;
}
